cmake_minimum_required(VERSION 3.22)
project(FallOfDutyNative LANGUAGES CXX)

include(CTest)
option(FOD_DESKTOP "Build the interactive SDL2 host application" ON)
find_package(ZLIB 1.2.9 REQUIRED)

add_library(fod_core app.cpp assets.cpp audio.cpp game.cpp hud.cpp light.cpp render.cpp trace.cpp)
target_link_libraries(fod_core PRIVATE ZLIB::ZLIB)
target_include_directories(fod_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(fod_core PUBLIC cxx_std_17)
if(MSVC)
    target_compile_options(fod_core PRIVATE /W4)
else()
    target_compile_options(fod_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(fod_host host/main.cpp)
target_link_libraries(fod_host PRIVATE fod_core)
if(FOD_DESKTOP)
    find_package(SDL2 CONFIG REQUIRED)
    target_sources(fod_host PRIVATE host/desktop.cpp)
    target_link_libraries(fod_host PRIVATE SDL2::SDL2)
    target_compile_definitions(fod_host PRIVATE FOD_DESKTOP=1)
endif()

set(FOD_TEST_PACK "${CMAKE_CURRENT_SOURCE_DIR}/assets/fall-of-duty.fodp" CACHE FILEPATH "Game pack for native tests")
if(BUILD_TESTING)
    add_executable(fod_tests host/tests.cpp)
    target_link_libraries(fod_tests PRIVATE fod_core)
    add_test(NAME native_tests COMMAND fod_tests "${FOD_TEST_PACK}")
    add_executable(fod_platform_tests host/platform_tests.cpp)
    target_link_libraries(fod_platform_tests PRIVATE fod_core)
    add_test(NAME platform_tests COMMAND fod_platform_tests "${FOD_TEST_PACK}")
endif()
